== (a) 運動方程式と特性方程式 (sympy) ==
  det[[λ²+K/p, -(b/p)λ], [(b/p)λ, λ²-K/p]] = -K**2/p**2 + b**2*lambda**2/p**2 + lambda**4
  w1 = sqrt(((b/p)² + V)/2)、w2 = (K/p)/w1、V = sqrt((b/p)⁴ + 4 (K/p)²)
  根: λ² = -w1² → det = 0、λ² = +w2² → det = 0
  w1² + w2² - V = 0、w1² - w2² - (b/p)² = 0、w1 w2 - K/p = 0

== (e) 固有モードと a, bw, c, dw の意味 ==
  λ = +i w1: Y/X = I*(w1 - w2)/sqrt(w1**2 - w2**2)  (wd = (b/p)/w+ = sqrt(w1**2 - w2**2)/(w1 + w2)。±i wd または ±1/wd と比べる: I、I*(w1 - w2)/(w1 + w2))
  λ = -i w1: Y/X = I*(-w1 + w2)/sqrt(w1**2 - w2**2)  (wd = (b/p)/w+ = sqrt(w1**2 - w2**2)/(w1 + w2)。±i wd または ±1/wd と比べる: -I、I*(-w1 + w2)/(w1 + w2))
  λ = +w2: Y/X = (w1 + w2)/sqrt(w1**2 - w2**2)  (wd = (b/p)/w+ = sqrt(w1**2 - w2**2)/(w1 + w2)。±i wd または ±1/wd と比べる: (w1 + w2)/(w1 - w2)、1)
  λ = -w2: Y/X = (-w1 - w2)/sqrt(w1**2 - w2**2)  (wd = (b/p)/w+ = sqrt(w1**2 - w2**2)/(w1 + w2)。±i wd または ±1/wd と比べる: (-w1 - w2)/(w1 - w2)、-1)
