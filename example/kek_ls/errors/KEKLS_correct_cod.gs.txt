 ! 固有値の大きい ie 個を使った疑似逆行列で COD を補正する。戻り値 {kick, 補正後の COD}
 CodCorr[resmat_,cod_,ie_]:=(
   tres        = resmat;
   res         = Transpose[tres];
   dim         = Length[res[1]];
   sqres       = tres.res;
   sqcod       = tres.cod;
   {lam, ivec} = Eigensystem[sqres];
   laminv      = Table[0.0, {i,1,dim}];
   lam = Re[lam];
   stdlam      = Sort[lam, Greater];
   lam = Abs[lam];
   Do[
     Do[
       If[stdlam[i]===lam[j],
         laminv[j]=1.0/stdlam[i];
       ];
     ,{j,dim}];
   ,{i, ie}];
   laminvmat   = DiagonalMatrix[laminv];
   ivect       = Transpose[ivec];
   invsqres    = ivect.laminvmat.ivec;
   kick        = -invsqres.sqcod;
   rescod      = cod + res.kick;
   out         = {kick, rescod};
   out
 );
 
 ! COD 応答行列 Responsex[j,i] (水平)、Responsey[j,i] (垂直): ステアリング j の単位キックに対するモニター i の COD
 CalculateCODResponse[]:=(
   FFS[cc];
   dtnx = LINE["NAME","HD*.*"];
   dtny = LINE["NAME","VD*.*"];
   dtbmx = Twiss["BX","PM.*"];
   dtbmy = Twiss["BY","PM.*"];
   dtbcx = Twiss["BX","HD*.*"];
   dtbcy = Twiss["BY","VD*.*"];
   dttmx = Twiss["NX","PM.*"];
   dttmy = Twiss["NY","PM.*"];
   dttcx = Twiss["NX","HD*.*"];
   dttcy = Twiss["NY","VD*.*"];
   ICX = Length[dtnx];
   ICY = Length[dtny];
   IM = Length[dtbmx];
   Print["ICX = ",ICX,"   ICY = ",ICY,"   IM = ",IM];
   Do[
     Do[
       Responsex[j,i] = 0.5*Sqrt[dtbmx[i]*dtbcx[j]]*Cos[tunex*Pi-Abs[dttmx[i]-dttcx[j]]]/Sin[tunex*Pi];
     ,{i,IM}];
   ,{j,ICX}];
   Do[
     Do[
       Responsey[j,i] = 0.5*Sqrt[dtbmy[i]*dtbcy[j]]*Cos[tuney*Pi-Abs[dttmy[i]-dttcy[j]]]/Sin[tuney*Pi];
     ,{i,IM}];
   ,{j,ICY}];
 );
