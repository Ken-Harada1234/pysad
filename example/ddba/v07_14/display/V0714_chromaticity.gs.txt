! V0714_chromaticity: δ に対するチューンの変化 (色収差)。六極 0 (自然クロマティシティ) と、六極 ON (最終設定)
!
! 元: calc_chromaticity.sad (7 つのリング RING・RINGN・RINGA・RINGL・RINGM・RINGLA・RINGMA で、六極を 0 にして、
!     δ = ±0.5% (RING) / ±1% (他) の 3 次の PolynomialFit の係数 [1,2] を出す)。amp_dep_ts_dp.sad (δ を 0〜±5% で振る) の δ スキャンも含める。
! 注意 (K64 のバグ B18): δ を ±対称に振ったデータを PolynomialFit に渡すと係数の次数補正が壊れる (docs/11 C6、docs/21 B18)。
!       元の台本が出していた「係数 [1,2]」は ξ そのものではない (δ の 1 次の係数 + 高次の混ざり。
!       出力の fit 係数と中心差分の ξ を並べて見比べる)。この台本は元の出し方 (係数 [1,2]) を残しつつ、正しい ξ を中心差分 ξ = [ν(+dδ) - ν(-dδ)]/(2 dδ)
!       (dδ = 1e-4) でも出す。
! 流れ: 各リングで六極 0 → δ スキャン → PolynomialFit → 係数 [1,2]。続いて最終設定の六極のままで同じ。

 ChromRing[ring_, dpm_, dst_]:=(
   FFS["USE "//ring];
   If[soff == 1, FFS["S* 0;"]];
   FFS["cell calc;"];
   FFS["calc;",6];
   FFS["emit;",6];
   tunex=Table[{i,0},{i,-dpm,dpm,dst}];
   tuney=Table[{i,0},{i,-dpm,dpm,dst}];
   j=0;
   Do[
     j = j + 1;
     DP0=i;
     FFS["cell calc;"];
     tunex[j,2] = Twiss["NX","$$$"]/2/Pi;
     tuney[j,2] = Twiss["NY","$$$"]/2/Pi;
   ,{i,-dpm,dpm,dst}];
   DP0 = 0;
   chrx = PolynomialFit[tunex,3];
   chry = PolynomialFit[tuney,3];
   ! 中心差分の ξ (dδ = 1e-4)
   DP0 = 1e-4;
   FFS["cell calc;"];
   nxp = Twiss["NX","$$$"]/2/Pi; nyp = Twiss["NY","$$$"]/2/Pi;
   DP0 = -1e-4;
   FFS["cell calc;"];
   nxm = Twiss["NX","$$$"]/2/Pi; nym = Twiss["NY","$$$"]/2/Pi;
   DP0 = 0;
   FFS["cell calc;"];
   xix = (nxp - nxm)/2e-4; xiy = (nyp - nym)/2e-4;
 );
