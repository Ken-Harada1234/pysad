 ! ---- リング全体の δ に対するチューン (6 極 0)。元の挙動のまま、fit の係数 [1,2] ----
 ! ChromFitRing[リング名, δ の範囲, 刻み] → chrx, chry (PolynomialFit の結果)、emitx (最後の δ での EMITX)
 ChromFitRing[ring_, dpm_, dst_]:=(
   FFS["USE "//ring,6];
   FFS["cell calc;",6];
   FFS["calc;",6];
   FFS["emit;",6];
   tunex=Table[{i,0},{i,-dpm,dpm,dst}];
   tuney=Table[{i,0},{i,-dpm,dpm,dst}];
   FFS["S* 0;"];
   j=0;
   Do[
     j = j + 1;
     DP0=i;
     FFS["cell calc emit;"];
     tunex[j,2] = Twiss["NX","$$$"]/2/Pi;
     tuney[j,2] = Twiss["NY","$$$"]/2/Pi;
   ,{i,-dpm,dpm,dst}];
   chrx = PolynomialFit[tunex,3];
   chry = PolynomialFit[tuney,3];
   emitx = EMITX;
 );
